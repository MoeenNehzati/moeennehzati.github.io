% Notes may contain characters like '&' that break moderncv's tabular-based layout.
% Sanitize them at print time (so `.bib` can contain e.g. `notes = {R&R at ...}`).
\ExplSyntaxOn
\cs_new_protected:Npn \cvbibSanitizeNote #1
  {
    \tl_set:Nn \l_tmpa_tl { #1 }
    \tl_replace_all:Nnn \l_tmpa_tl { & } { \& }
    \tl_use:N \l_tmpa_tl
  }
\ExplSyntaxOff
\DeclareFieldFormat{note}{\cvbibSanitizeNote{#1}}
\DeclareSourcemap{
  \maps[datatype=bibtex]{
    % `abbr` is a nonstandard field; map it into `shortjournal` (a standard biblatex field)
    % so it survives biber and we can reliably print it.
    \map{
      \step[fieldsource=abbr, fieldtarget=shortjournal, final]
      \step[fieldset=abbr, null]
    }
    % Some entries may store notes in a nonstandard `extra` field; map it into `note`.
    \map{
      \step[fieldsource=extra, fieldtarget=note, final]
      \step[fieldset=extra, null]
    }
    % Some entries may use `notes` (plural); map it into `note` (biblatex standard).
    \map{
      \step[fieldsource=notes, fieldtarget=note, final]
      \step[fieldset=notes, null]
    }
    % `credit_url`/`credit_text` (optional, on `software` entries: a link to a
    % paper the software implements) are nonstandard; map them into biblatex's
    % generic `usera`/`userb` fields so they survive biber.
    \map{
      \step[fieldsource=credit_url, fieldtarget=usera, final]
      \step[fieldset=credit_url, null]
    }
    \map{
      \step[fieldsource=credit_text, fieldtarget=userb, final]
      \step[fieldset=credit_text, null]
    }
  }
}

\defbibcheck{published}{%
  \ifkeyword{published}{% keep entry
  }{\skipentry}% else skip
}

  \ifkeyword{preprint}{% keep entry
  }{\skipentry}% else skip
}
\defbibcheck{wip}{%
  \ifkeyword{work-in-progress}{% keep entry
  }{\skipentry}% else skip
}

% Render bibliography entries as moderncv `\cventry` lines
% Format: year · title · (linked abbr, if provided) · authors
\newcommand*{\cvbibvenue}{%
  \iffieldundef{shortjournal}{%
    \iflistundef{institution}{}{\printlist{institution}}%
  }{%
    \iffieldundef{url}{%
      \printfield{shortjournal}%
    }{%
      \href{\strfield{url}}{\printfield{shortjournal}}%
    }%
  }%
}

\newbool{cvbibprinted}
\newcommand*{\cvbibdetails}{%
  \boolfalse{cvbibprinted}%
  % Venue (linked when url exists)
  \iffieldundef{shortjournal}{%
    \iflistundef{institution}{}{%
      \printlist{institution}%
      \booltrue{cvbibprinted}%
    }%
  }{%
    \cvbibvenue%
    \booltrue{cvbibprinted}%
  }%
  % Optional note/extra
  \iffieldundef{note}{}{%
    \ifbool{cvbibprinted}{\addcomma\space}{}%
    \printfield{note}%
    \booltrue{cvbibprinted}%
  }%
  % Authors
  \ifnameundef{author}{}{%
    \ifbool{cvbibprinted}{\addcomma\space}{}%
    \printnames{author}%
  }%
}

\newcommand*{\cvbibdriver}{%
  % moderncv's \cventry prints fields 3--6 inline (comma-separated) and field 7 on a new line.
  % Keep venue + optional note + authors on a single line.
  \cventry{\printfield{year}}{\printfield{title}}{\cvbibdetails}{}{}{}{}%
  \finentry
}
